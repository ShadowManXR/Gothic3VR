function(upscale_shader name source profile entry define)
  add_custom_command(OUTPUT "${UPSCALE_GENERATED}/${name}.h"
    COMMAND "${G3VR_FXC}" /nologo /O3 /T ${profile} /E ${entry} /D ${define}
      /I "${CMAKE_CURRENT_SOURCE_DIR}/third_party/upscalers/nis" /I "${CMAKE_CURRENT_SOURCE_DIR}/third_party/upscalers/cas"
      /I "${CMAKE_CURRENT_SOURCE_DIR}/third_party/upscalers/fxaa" /I "${CMAKE_CURRENT_SOURCE_DIR}/third_party/upscalers/smaa"
      /Fh "${UPSCALE_GENERATED}/${name}.h" /Vn ${name} "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/${source}"
    DEPENDS "src/shaders/${source}" ${UPSCALE_HEADERS} VERBATIM)
  target_sources(Gothic3VRUpscaler PRIVATE "${UPSCALE_GENERATED}/${name}.h")
endfunction()
