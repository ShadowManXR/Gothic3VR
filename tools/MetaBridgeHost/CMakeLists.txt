cmake_minimum_required(VERSION 3.24)
project(Gothic3VRMetaHost LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "The Meta compositor host must be built for x64")
endif()
add_executable(Gothic3VRMetaHost64 main.cpp MetaHost.manifest)
target_compile_features(Gothic3VRMetaHost64 PRIVATE cxx_std_17)
target_compile_definitions(Gothic3VRMetaHost64 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(Gothic3VRMetaHost64 PRIVATE /EHsc /O2 /W4)
target_include_directories(Gothic3VRMetaHost64 PRIVATE ../../src ../../third_party/openxr/include)
target_link_libraries(Gothic3VRMetaHost64 PRIVATE d3d11 dxgi d3dcompiler advapi32)
add_custom_command(TARGET Gothic3VRMetaHost64 POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/openxr/x64/openxr_loader.dll" "$<TARGET_FILE_DIR:Gothic3VRMetaHost64>/openxr_loader64.dll"
 COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/openxr/LICENSE" "$<TARGET_FILE_DIR:Gothic3VRMetaHost64>/OPENXR-META64-LICENSE.txt")
option(G3VR_META_GPU_FIXTURE "Test-only CPU validation of cross-process XR pixels; never install" OFF)
add_executable(DiagnosticMetaPipe64Tests EXCLUDE_FROM_ALL ../../tests/MetaPipeTests.cpp)
target_compile_features(DiagnosticMetaPipe64Tests PRIVATE cxx_std_17)
target_compile_definitions(DiagnosticMetaPipe64Tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(DiagnosticMetaPipe64Tests PRIVATE /EHsc /O2)
target_include_directories(DiagnosticMetaPipe64Tests PRIVATE ../../src)
if(G3VR_META_GPU_FIXTURE)
 target_compile_definitions(Gothic3VRMetaHost64 PRIVATE G3VR_META_GPU_FIXTURE)
endif()
add_executable(DiagnosticMetaBridgeWire64Tests EXCLUDE_FROM_ALL ../../tests/MetaBridgeWireTests.cpp)
target_compile_features(DiagnosticMetaBridgeWire64Tests PRIVATE cxx_std_17)
target_compile_definitions(DiagnosticMetaBridgeWire64Tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(DiagnosticMetaBridgeWire64Tests PRIVATE /EHsc /O2)
target_include_directories(DiagnosticMetaBridgeWire64Tests PRIVATE ../../src ../../third_party/openxr/include)
add_executable(DiagnosticMetaFrameTimingTests EXCLUDE_FROM_ALL ../../tests/MetaFrameTimingTests.cpp)
target_compile_features(DiagnosticMetaFrameTimingTests PRIVATE cxx_std_17)
target_compile_options(DiagnosticMetaFrameTimingTests PRIVATE /EHsc /O2)
target_include_directories(DiagnosticMetaFrameTimingTests PRIVATE ../../src)
add_executable(DiagnosticMetaFrameWaiterTests EXCLUDE_FROM_ALL ../../tests/MetaFrameWaiterTests.cpp)

target_compile_definitions(DiagnosticMetaFrameWaiterTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
